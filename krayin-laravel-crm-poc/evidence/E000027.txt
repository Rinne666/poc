186:             $stage = $this->stageRepository->findOrFail($data['lead_pipeline_stage_id']);

189:         } else {

195:                 $pipeline = $this->pipelineRepository->findOrFail($data['lead_pipeline_id']);

198:             $stage = $pipeline->stages()->first();
199:
200:             $data['lead_pipeline_stage_id'] = $stage->id;
201:         }
202:
203:         if (in_array($stage->code, ['won', 'lost'])) {
204:             $data['closed_at'] = Carbon::now();
205:         }
206:
207:         $lead = $this->leadRepository->create($data);
208:
209:         Event::dispatch('lead.create.after', $lead);
210:
211:         if (request()->ajax()) {
212:             return response()->json([
213:                 'message' => trans('admin::app.leads.create-success'),
214:                 'data' => new LeadResource($lead),
215:             ]);
216:         }
217:
218:         session()->flash('success', trans('admin::app.leads.create-success'));
219:
220:         if (! empty($data['lead_pipeline_id'])) {
